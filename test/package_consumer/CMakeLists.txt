# Standalone archive consumer. This file is copied beside the renamed extracted library.
cmake_minimum_required(VERSION 3.16)
project(TUSB32xPackageConsumer LANGUAGES CXX)

# Check before CACHE PATH can normalize a caller-supplied relative value.
if(DEFINED TUSB32X_PACKAGE_ROOT AND NOT IS_ABSOLUTE "${TUSB32X_PACKAGE_ROOT}")
  message(FATAL_ERROR "TUSB32X_PACKAGE_ROOT must be supplied as an absolute path")
endif()
set(TUSB32X_PACKAGE_ROOT "" CACHE PATH "Absolute path to the renamed extracted library")
if(NOT IS_ABSOLUTE "${TUSB32X_PACKAGE_ROOT}" OR
   NOT EXISTS "${TUSB32X_PACKAGE_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "TUSB32X_PACKAGE_ROOT must name an absolute extracted library root")
endif()
get_filename_component(package_root "${TUSB32X_PACKAGE_ROOT}" REALPATH)
add_subdirectory("${package_root}" "${CMAKE_CURRENT_BINARY_DIR}/core")
if(TUSB32X_BUILD_TESTS)
  message(FATAL_ERROR "Package consumption requires the library's default tests-OFF build")
endif()

# Fail if a changed build file silently substitutes a checkout or sibling implementation.
get_target_property(core_source_dir TUSB32x SOURCE_DIR)
get_filename_component(core_source_dir "${core_source_dir}" REALPATH)
if(NOT core_source_dir STREQUAL package_root)
  message(FATAL_ERROR "TUSB32x target did not originate in the extracted library")
endif()
foreach(property IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
  get_target_property(core_includes TUSB32x ${property})
  if(NOT core_includes STREQUAL "${package_root}/include")
    message(FATAL_ERROR "Unexpected ${property}: ${core_includes}")
  endif()
endforeach()
get_target_property(core_sources TUSB32x SOURCES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/package-consumer-origin.txt" "${package_root}\n")
foreach(source IN LISTS core_sources)
  get_filename_component(source_path "${source}" REALPATH BASE_DIR "${core_source_dir}")
  string(FIND "${source_path}" "${package_root}/src/" source_prefix)
  if(NOT source_prefix EQUAL 0 OR NOT EXISTS "${source_path}")
    message(FATAL_ERROR "Core source escaped extracted src directory: ${source_path}")
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/package-consumer-origin.txt" "${source_path}\n")
endforeach()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/package-consumer-origin.txt" "${package_root}/include\n")

add_executable(tusb32x_package_consumer main.cpp)
target_link_libraries(tusb32x_package_consumer PRIVATE TUSB32x)
set_target_properties(tusb32x_package_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(tusb32x_package_consumer PRIVATE /W4 /WX /permissive-)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(tusb32x_package_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()
enable_testing()
add_test(NAME tusb32x_package_consumer COMMAND tusb32x_package_consumer)
set_tests_properties(tusb32x_package_consumer PROPERTIES TIMEOUT 10)
